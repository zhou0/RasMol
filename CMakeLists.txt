option(RASMOL_AUTO_INSTALL "Automatically install missing dependencies" OFF)

if(RASMOL_AUTO_INSTALL)
    set(NEEDS_INSTALL FALSE)

    # Check for Bison 2.7+
    find_package(BISON 2.7 QUIET)
    if(NOT BISON_FOUND AND NOT BISON_EXECUTABLE)
        set(NEEDS_INSTALL TRUE)
        message(STATUS "Bison 2.7+ not found.")
    endif()

    # Check for Tcl/Tk
    find_package(TCL QUIET)
    find_package(TK QUIET)
    if(NOT TCL_FOUND OR NOT TK_FOUND)
        if(NOT WIN32)
            set(NEEDS_INSTALL TRUE)
            message(STATUS "Tcl/Tk not found.")
        endif()
    endif()

    # Check for HDF5
    if(APPLE)
        find_path(HDF5_INCLUDE_DIR hdf5.h PATHS /opt/local/include /opt/homebrew/opt/hdf5/include /usr/local/opt/hdf5/include QUIET)
        if(NOT HDF5_INCLUDE_DIR)
            set(NEEDS_INSTALL TRUE)
            message(STATUS "HDF5 not found.")
        endif()
    else()
        find_package(HDF5 QUIET COMPONENTS C)
        if(NOT HDF5_FOUND)
            find_path(HDF5_INCLUDE_DIR hdf5.h PATH_SUFFIXES hdf5/serial QUIET)
            if(NOT HDF5_INCLUDE_DIR)
                set(NEEDS_INSTALL TRUE)
                message(STATUS "HDF5 not found.")
            endif()
        endif()
    endif()

    # Check for VTK 9.1.0+
    find_package(VTK 9.1.0 QUIET)
    if(NOT VTK_FOUND)
        set(NEEDS_INSTALL TRUE)
        message(STATUS "VTK 9.1.0+ not found.")
    endif()

    # Check for libaec (SZIP)
    find_path(LIBAEC_INCLUDE_DIR libaec.h PATHS /opt/local/include /usr/local/include /usr/include QUIET)
    find_library(LIBAEC_LIBRARY NAMES aec libaec PATHS /opt/local/lib /usr/local/lib /usr/lib QUIET)
    find_library(SZ_LIBRARY NAMES sz linszaec szip libsz PATHS /opt/local/lib /usr/local/lib /usr/lib QUIET)
    if(NOT LIBAEC_INCLUDE_DIR OR NOT LIBAEC_LIBRARY OR NOT SZ_LIBRARY)
        set(NEEDS_INSTALL TRUE)
        message(STATUS "libaec/szip not found.")
    endif()

    # Check for zlib
    find_package(ZLIB QUIET)
    if(NOT ZLIB_FOUND)
        set(NEEDS_INSTALL TRUE)
        message(STATUS "zlib not found.")
    endif()

    if(NEEDS_INSTALL)
        message(STATUS "Dependencies missing, attempting auto-installation...")
        if(APPLE)
            message(STATUS "Installing dependencies via MacPorts...")
            execute_process(COMMAND sudo port selfupdate)
            set(PORT_ARGS "")
            if(CMAKE_OSX_ARCHITECTURES)
                list(GET CMAKE_OSX_ARCHITECTURES 0 ARCH)
                set(PORT_ARGS "configure.build_arch=${ARCH}" "configure.universal_archs=${ARCH}")
            endif()
            execute_process(COMMAND sudo port install bison)
            execute_process(COMMAND sudo port install tcl tk libjpeg-turbo hdf5 zstd webp lerc libdeflate jbigkit xz zlib libaec vtk ${PORT_ARGS})
        elseif(UNIX AND NOT APPLE)
            find_program(APT_GET apt-get)
            find_program(DNF dnf)
            find_program(PACMAN pacman)
            if(APT_GET)
                message(STATUS "Installing dependencies via apt...")
                execute_process(COMMAND sudo apt-get update)
                execute_process(COMMAND sudo apt-get install -y rpm libx11-dev libxext-dev libxi-dev libhdf5-dev tcl-dev tk-dev bison libxpm-dev libdeflate-dev libjbig-dev liblerc-dev libwebp-dev libjpeg-dev imagemagick libvtk9-dev libaec-dev zlib1g-dev)
            elseif(DNF)
                message(STATUS "Installing dependencies via dnf...")
                execute_process(COMMAND sudo dnf install -y rpm-build libX11-devel libXext-devel libXi-devel hdf5-devel tcl-devel tk-devel bison libXpm-devel libdeflate-devel jbigkit-devel lerc-devel libwebp-devel libjpeg-turbo-devel ImageMagick vtk-devel libaec-devel zlib-devel)
            elseif(PACMAN)
                message(STATUS "Installing dependencies via pacman...")
                execute_process(COMMAND sudo pacman -Sy --noconfirm rpm-tools libx11 libxext libxi hdf5 tcl tk bison libxpm libdeflate jbigkit lerc libwebp libjpeg-turbo imagemagick vtk libaec zlib)
            else()
                message(WARNING "No supported package manager (apt, dnf, pacman) found.")
            endif()
        elseif(WIN32)
            message(STATUS "Installing dependencies on Windows...")

            # Bison download for Windows
            if(NOT BISON_FOUND AND NOT BISON_EXECUTABLE)
                message(STATUS "Downloading win_flex_bison...")
                set(BISON_ZIP_URL "https://github.com/lexxmark/winflexbison/releases/download/v2.4/win_flex_bison-2.4.zip")
                set(BISON_ZIP_DIR "${CMAKE_BINARY_DIR}/win_flex_bison")
                file(MAKE_DIRECTORY "${BISON_ZIP_DIR}")
                file(DOWNLOAD "${BISON_ZIP_URL}" "${BISON_ZIP_DIR}/win_flex_bison.zip" SHOW_PROGRESS)
                execute_process(COMMAND ${CMAKE_COMMAND} -E tar xf "${BISON_ZIP_DIR}/win_flex_bison.zip" WORKING_DIRECTORY "${BISON_ZIP_DIR}")
                set(BISON_EXECUTABLE "${BISON_ZIP_DIR}/win_bison.exe" CACHE FILEPATH "Path to bison executable" FORCE)
                set(FLEX_EXECUTABLE "${BISON_ZIP_DIR}/win_flex.exe" CACHE FILEPATH "Path to flex executable" FORCE)
                message(STATUS "Bison set to ${BISON_EXECUTABLE}")
            endif()

            execute_process(COMMAND choco --version RESULT_VARIABLE CHOCO_CHECK OUTPUT_QUIET ERROR_QUIET)
            if(CHOCO_CHECK EQUAL 0)
                execute_process(COMMAND choco install magicsplat-tcl-tk --version=1.16.0 -y)
            else()
                message(WARNING "Chocolatey not found. Cannot install Tcl/Tk automatically.")
            endif()

            # Install VTK via vcpkg on Windows
            execute_process(COMMAND vcpkg --version RESULT_VARIABLE VCPKG_CHECK OUTPUT_QUIET ERROR_QUIET)
            if(VCPKG_CHECK EQUAL 0)
                execute_process(COMMAND vcpkg install vtk:x64-windows libaec:x64-windows zlib:x64-windows)
                # vcpkg often needs some help to be found if not using the toolchain file
                set(VCPKG_ROOT "C:/vcpkg" CACHE PATH "Vcpkg root directory")
                if(EXISTS "${VCPKG_ROOT}")
                   list(APPEND CMAKE_PREFIX_PATH "${VCPKG_ROOT}/installed/x64-windows")
                endif()
            else()
                message(WARNING "vcpkg not found. Cannot install VTK automatically.")
            endif()

            execute_process(COMMAND dotnet --version RESULT_VARIABLE DOTNET_CHECK OUTPUT_QUIET ERROR_QUIET)
            if(DOTNET_CHECK EQUAL 0)
                execute_process(COMMAND wix --version RESULT_VARIABLE WIX_CHECK OUTPUT_QUIET ERROR_QUIET)
                if(NOT WIX_CHECK EQUAL 0)
                    execute_process(COMMAND dotnet tool install --global WiX)
                endif()
            else()
                message(WARNING "dotnet not found. Cannot install WiX automatically.")
            endif()
        endif()
    else()
        message(STATUS "All basic dependencies found. Skipping auto-installation.")
    endif()
endif()

if(NOT CQRLIB_LIBRARY OR CQRLIB_LIBRARY STREQUAL "CQRlib")
    fetchcontent_declare(cqrlib
        GIT_REPOSITORY https://github.com/zhou0/cqrlib.git
        GIT_TAG master
    )
    fetchcontent_getproperties(cqrlib)
    if(NOT cqrlib_POPULATED)
        fetchcontent_populate(cqrlib)
        # Force shared library for CQRlib
        file(READ ${cqrlib_SOURCE_DIR}/CMakeLists.txt CQR_CMAKE)
        string(REPLACE "add_library(CQRlib STATIC" "add_library(CQRlib SHARED" CQR_CMAKE "${CQR_CMAKE}")
        file(WRITE ${cqrlib_SOURCE_DIR}/CMakeLists.txt "${CQR_CMAKE}")
    endif()
    add_subdirectory(${cqrlib_SOURCE_DIR} ${cqrlib_BINARY_DIR} EXCLUDE_FROM_ALL)
    set(CQRLIB_LIBRARY CQRlib CACHE STRING "" FORCE)
    set(CQRLIB_INCLUDE_DIR ${cqrlib_SOURCE_DIR} CACHE PATH "" FORCE)
endif()


if(NOT CVECTOR_LIBRARY OR CVECTOR_LIBRARY STREQUAL "CV_shared")
    fetchcontent_declare(cvector
        GIT_REPOSITORY https://github.com/zhou0/CVector.git
        GIT_TAG master
    )
    fetchcontent_getproperties(cvector)
    if(NOT cvector_POPULATED)
        fetchcontent_populate(cvector)
        # Patch CVector/CMakeLists.txt to remove its own output directory overrides
        file(READ ${cvector_SOURCE_DIR}/CMakeLists.txt CV_CMAKE)
        string(REPLACE "set(CMAKE_RUNTIME_OUTPUT_DIRECTORY" "#set(CMAKE_RUNTIME_OUTPUT_DIRECTORY" CV_CMAKE "${CV_CMAKE}")
        string(REPLACE "set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY" "#set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY" CV_CMAKE "${CV_CMAKE}")
        string(REPLACE "set(CMAKE_LIBRARY_OUTPUT_DIRECTORY" "#set(CMAKE_LIBRARY_OUTPUT_DIRECTORY" CV_CMAKE "${CV_CMAKE}")
        # Resolve MSVC duplicate .lib name conflict between CV_static and CV_shared
        string(REPLACE "set_target_properties(CV_static PROPERTIES OUTPUT_NAME CVector)" "set_target_properties(CV_static PROPERTIES OUTPUT_NAME CVectorStatic)" CV_CMAKE "${CV_CMAKE}")
        string(REPLACE "set_target_properties(CV_static PROPERTIES OUTPUT_NAME \"CVector\")" "set_target_properties(CV_static PROPERTIES OUTPUT_NAME \"CVectorStatic\")" CV_CMAKE "${CV_CMAKE}")
        # Ensure static and shared targets have unique output names on Windows to avoid rule conflicts
        string(REPLACE "set_target_properties(CV_static PROPERTIES OUTPUT_NAME CVector)" "set_target_properties(CV_static PROPERTIES OUTPUT_NAME CVectorStatic)" CV_CMAKE "${CV_CMAKE}")
        string(REPLACE "set_target_properties(CV_static PROPERTIES OUTPUT_NAME \"CVector\")" "set_target_properties(CV_static PROPERTIES OUTPUT_NAME \"CVectorStatic\")" CV_CMAKE "${CV_CMAKE}")
        # Disable installation in subproject to avoid more conflicts
        string(APPEND CV_CMAKE "\nset(CV_INSTALL OFF CACHE BOOL \"\" FORCE)\n")
        file(WRITE ${cvector_SOURCE_DIR}/CMakeLists.txt "${CV_CMAKE}")
    endif()
    add_subdirectory(${cvector_SOURCE_DIR} ${cvector_BINARY_DIR} EXCLUDE_FROM_ALL)
    set(CVECTOR_LIBRARY CV_shared CACHE STRING "" FORCE)
    set(CVECTOR_INCLUDE_DIR ${cvector_SOURCE_DIR} CACHE PATH "" FORCE)
endif()

# Tcl/Tk Support
if(APPLE)
    # macOS Homebrew and framework logic hints
    set(TCL_TK_PATHS /opt/local /opt/homebrew/opt/tcl-tk /usr/local/opt/tcl-tk)
    set(CMAKE_FRAMEWORK_PATH /Library/Frameworks /System/Library/Frameworks)
    find_library(TCL_LIBRARY NAMES tcl9.0 tcl9 tcl8.6 tcl86 tcl HINTS ${TCL_TK_PATHS} PATH_SUFFIXES lib)
    find_library(TK_LIBRARY NAMES tk9.0 tk9 tk8.6 tk86 tk HINTS ${TCL_TK_PATHS} PATH_SUFFIXES lib)
    find_path(TCL_INCLUDE_PATH tcl.h HINTS ${TCL_TK_PATHS} PATH_SUFFIXES include Tcl.framework/Headers)
    find_path(TK_INCLUDE_PATH tk.h HINTS ${TCL_TK_PATHS} PATH_SUFFIXES include Tk.framework/Headers)
endif()
if(NOT TCL_LIBRARY OR NOT TK_LIBRARY)
    find_package(TCL QUIET)
    find_package(TK QUIET)
endif()

if(NOT TCL_INCLUDE_PATH OR NOT TK_INCLUDE_PATH OR NOT TCL_LIBRARY OR NOT TK_LIBRARY)
    if(WIN32)
        set(TCL_NAMES tcl86 tcl86t)
        set(TK_NAMES tk86 tk86t)
    else()
        set(TCL_NAMES tcl8.6 tcl86)
        set(TK_NAMES tk8.6 tk86)
    endif()

    find_path(TCL_INCLUDE_PATH tcl.h PATH_SUFFIXES tcl8.6 tcl86 tcl)
    find_path(TK_INCLUDE_PATH tk.h PATH_SUFFIXES tk8.6 tk86 tk)
    find_library(TCL_LIBRARY NAMES ${TCL_NAMES})
    find_library(TK_LIBRARY NAMES ${TK_NAMES})
endif()

if(TCL_LIBRARY AND TK_LIBRARY)
    set(RASMOL_TCL_SOURCES warpings/tcl/rasmol_tcl.c ${CORE_SOURCES})
    if(VTK_FOUND)
        list(APPEND RASMOL_TCL_SOURCES warpings/tcl/rasmol_vtk.cpp)
    endif()
    if(APPLE)
        add_executable(rasmol_tcl MACOSX_BUNDLE ${RASMOL_TCL_SOURCES})
        set_target_properties(rasmol_tcl PROPERTIES
            OUTPUT_NAME "RasMol"
            INSTALL_RPATH "@executable_path/../Frameworks"
        )
        set_target_properties(rasmol_tcl PROPERTIES
            MACOSX_BUNDLE_BUNDLE_NAME "RasMol"
            MACOSX_BUNDLE_GUI_IDENTIFIER "org.openrasmol.rasmol"
            MACOSX_BUNDLE_BUNDLE_VERSION "${CPACK_PACKAGE_VERSION}"
            MACOSX_BUNDLE_SHORT_VERSION_STRING "${CPACK_PACKAGE_VERSION}"
            MACOSX_BUNDLE_ICON_FILE "rasmol.icns"
        )
    else()
        add_executable(rasmol_tcl ${RASMOL_TCL_SOURCES})
    endif()
    target_include_directories(rasmol_tcl PRIVATE src ${TCL_INCLUDE_PATH} ${TK_INCLUDE_PATH})
    target_link_libraries(rasmol_tcl ${TCL_LIBRARY} ${TK_LIBRARY} ${EXTERNAL_LIBRARIES})
    if(VTK_FOUND)
        target_link_libraries(rasmol_tcl PRIVATE ${VTK_LIBRARIES})
        vtk_module_autoinit(TARGETS rasmol_tcl MODULES ${VTK_LIBRARIES})
    endif()
    if(UNIX AND NOT APPLE AND WITH_X11)
        target_link_libraries(rasmol_tcl ${X11_LIBRARIES})
    endif()
    file(GLOB TCL_SCRIPTS "gui/tcltk/*.tcl")
    if(APPLE)
        target_link_libraries(rasmol_tcl "-framework Cocoa" "-framework IOKit" "-framework CoreFoundation" "-framework CoreGraphics")
        target_compile_definitions(rasmol_tcl PRIVATE MAC_OSX_TK)

        # Add icons and resources to the bundle
        if(EXISTS "${CMAKE_BINARY_DIR}/assets/rasmol.icns")
            set_source_files_properties("${CMAKE_BINARY_DIR}/assets/rasmol.icns" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
            target_sources(rasmol_tcl PRIVATE "${CMAKE_BINARY_DIR}/assets/rasmol.icns")
        endif()
        set(BUNDLE_RESOURCES
            ${TCL_SCRIPTS}
            doc/rasmol.hlp
        )
        set_source_files_properties(${BUNDLE_RESOURCES} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
        target_sources(rasmol_tcl PRIVATE ${BUNDLE_RESOURCES})
    endif()
    if(APPLE)
        install(TARGETS rasmol_tcl BUNDLE DESTINATION .)
        # Install third-party shared libraries into the bundle
        set(ALL_LIBS ${EXTERNAL_LIBRARIES} ${TCL_LIBRARY} ${TK_LIBRARY} ${VTK_LIBRARIES})
        if(ALL_LIBS)
            list(REMOVE_DUPLICATES ALL_LIBS)
            foreach(lib ${ALL_LIBS})
                if(TARGET "${lib}")
                    get_target_property(LIB_TYPE "${lib}" TYPE)
                    if(LIB_TYPE STREQUAL "SHARED_LIBRARY")
                        install(TARGETS "${lib}" LIBRARY DESTINATION ${RASMOL_FRAMEWORKS_DIR})
                    endif()
                elseif(EXISTS "${lib}")
                    if(lib MATCHES "\\.dylib$" OR lib MATCHES "\\.framework$")
                        if(NOT lib MATCHES "^/usr/lib" AND NOT lib MATCHES "^/System/Library")
                            if(IS_DIRECTORY "${lib}")
                                install(DIRECTORY "${lib}" DESTINATION ${RASMOL_FRAMEWORKS_DIR})
                            else()
                                get_filename_component(LIB_NAME "${lib}" NAME)
                                get_filename_component(LIB_REAL_PATH "${lib}" REALPATH)
                                get_filename_component(LIB_REAL_NAME "${LIB_REAL_PATH}" NAME)
                                install(FILES "${LIB_REAL_PATH}" DESTINATION ${RASMOL_FRAMEWORKS_DIR})
                                if(NOT "${LIB_NAME}" STREQUAL "${LIB_REAL_NAME}")
                                     install(CODE "execute_process(COMMAND \${CMAKE_COMMAND} -E create_symlink \"\${LIB_REAL_NAME}\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/\${RASMOL_FRAMEWORKS_DIR}/\${LIB_NAME}\")")
                                endif()
                            endif()
                        endif()
                    endif()
                endif()
            endforeach()
        endif()
    else()
        install(TARGETS rasmol_tcl DESTINATION ${RASMOL_BIN_DIR})
    endif()
    if(NOT APPLE)
        install(FILES ${TCL_SCRIPTS} DESTINATION ${RASMOL_SHARE_DIR})
    endif()
else()
    message(WARNING "Tcl/Tk dependencies not fully found. rasmol_tcl target will not be created.")
    message(STATUS "TCL_INCLUDE_PATH: ${TCL_INCLUDE_PATH}")
    message(STATUS "TK_INCLUDE_PATH: ${TK_INCLUDE_PATH}")
    message(STATUS "TCL_LIBRARY: ${TCL_LIBRARY}")
    message(STATUS "TK_LIBRARY: ${TK_LIBRARY}")
endif()
